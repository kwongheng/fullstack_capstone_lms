CREATE TABLE USER (
    id            INT PRIMARY KEY AUTO_INCREMENT,
    email         VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name     VARCHAR(255) NOT NULL,
    phone         VARCHAR(20) NOT NULL,
    address       TEXT,
    role          ENUM('Admin', 'Member') DEFAULT 'Member' NOT NULL,

    -- Phone: must contain only digits AND cannot start with 0
    CONSTRAINT CHK_phone_digits_only 
        CHECK (phone REGEXP '^[0-9]+$'),
    CONSTRAINT CHK_phone_no_leading_zero 
        CHECK (phone NOT LIKE '0%')
);

CREATE TABLE MEMBER (
    user_id     INT PRIMARY KEY,
    member_id   VARCHAR(20) UNIQUE NOT NULL,
    join_date   DATE DEFAULT (CURRENT_DATE) NOT NULL,
    status      ENUM('Active', 'Suspended', 'Expired') DEFAULT 'Active' NOT NULL,
    
    CONSTRAINT FK_MEMBER_USER 
        FOREIGN KEY (user_id) REFERENCES USER(id) 
        ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE BOOK (
    id               INT PRIMARY KEY AUTO_INCREMENT,
    isbn             VARCHAR(13) UNIQUE NOT NULL,
    title            VARCHAR(500) NOT NULL,
    author           VARCHAR(255) NOT NULL,
    category         VARCHAR(100),
    publication_year YEAR NOT NULL,           -- Proper YEAR data type (0000–9999)
    publisher        VARCHAR(255),
    total_copies     INT NOT NULL CHECK (total_copies >= 0),
    available_copies INT NOT NULL CHECK (available_copies >= 0),
    
    CONSTRAINT CHK_available_copies 
        CHECK (available_copies <= total_copies)
    
);

-- The rest of the tables remain exactly the same
CREATE TABLE BORROW (
    id            INT PRIMARY KEY AUTO_INCREMENT,
    user_id       INT NOT NULL,
    book_id       INT NOT NULL,
    borrow_date   DATETIME DEFAULT CURRENT_TIMESTAMP NOT NULL,
    due_date      DATETIME GENERATED ALWAYS AS (borrow_date + INTERVAL 14 DAY) STORED,
    return_date   DATETIME NULL,
    fine_amount   DECIMAL(6,2) DEFAULT 0.00 CHECK (fine_amount <= 20.00),
    times_renew   TINYINT DEFAULT 0 CHECK (times_renew <= 2),
    is_returned   BOOLEAN DEFAULT FALSE NOT NULL,
    
    CONSTRAINT FK_BORROW_MEMBER 
        FOREIGN KEY (user_id) REFERENCES MEMBER(user_id) 
        ON DELETE RESTRICT ON UPDATE CASCADE,
    CONSTRAINT FK_BORROW_BOOK 
        FOREIGN KEY (book_id) REFERENCES BOOK(id) 
        ON DELETE RESTRICT ON UPDATE CASCADE,
    CONSTRAINT CHK_return_date 
        CHECK (return_date IS NULL OR return_date >= borrow_date),
    CONSTRAINT UQ_USER_BOOK_ACTIVE 
        UNIQUE (user_id, book_id, is_returned)
);

CREATE TABLE RESERVATION (
    id                INT PRIMARY KEY AUTO_INCREMENT,
    user_id           INT NOT NULL,
    book_id           INT NOT NULL,
    reservation_date  DATETIME DEFAULT CURRENT_TIMESTAMP NOT NULL,
    expiry_date       DATETIME GENERATED ALWAYS AS (reservation_date + INTERVAL 14 DAY) STORED,
    is_fulfilled      BOOLEAN DEFAULT FALSE NOT NULL,
    
    CONSTRAINT FK_RESERVE_MEMBER 
        FOREIGN KEY (user_id) REFERENCES MEMBER(user_id) 
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT FK_RESERVE_BOOK 
        FOREIGN KEY (book_id) REFERENCES BOOK(id) 
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT UQ_ACTIVE_RESERVATION 
        UNIQUE (user_id, book_id, is_fulfilled)
);

CREATE TABLE FINE (
    id         INT PRIMARY KEY AUTO_INCREMENT,
    borrow_id  INT NOT NULL,
    amount     DECIMAL(6,2) NOT NULL CHECK (amount <= 20.00),
    is_paid    BOOLEAN DEFAULT FALSE NOT NULL,
    paid_at    DATETIME NULL,
    
    CONSTRAINT FK_FINE_BORROW 
        FOREIGN KEY (borrow_id) REFERENCES BORROW(id) 
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT CHK_paid_at 
        CHECK (is_paid = FALSE OR paid_at IS NOT NULL)
);

-- Indexes
CREATE INDEX IDX_BOOK_TITLE ON BOOK(title);
CREATE INDEX IDX_BOOK_ISBN  ON BOOK(isbn);
CREATE INDEX IDX_BORROW_DATES ON BORROW(borrow_date, due_date);
