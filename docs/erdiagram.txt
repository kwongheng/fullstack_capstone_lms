erDiagram
  MEMBER||--o{BORROW:"borrows"
  MEMBER||--o{RESERVATION:"reserves"
  USER||--||MEMBER:"is-a"
  BOOK||--o{BORROW:"is borrowed"
  BOOK||--o{RESERVATION:"is reserved"
  BORROW||--o|FINE:"generates"

  USER {
    int id PK
    string email UK
    string password_hash
    string full_name
    string phone "only contains digits"
    string address
    string role "Admin / Member"
  }

  MEMBER {
    int user_id PK,FK
    string member_id UK "MEM-0001"
    date join_date
    string status "Active / Suspended / Expired"
  }

  BOOK {
    int id PK
    string isbn UK
    string title
    string author
    string category
    int publication_year
    string publisher
    int total_copies
    int available_copies "<= total_copies"
  }

  BORROW {
    int id PK
    int user_id FK
    int book_id FK
    datetime borrow_date
    datetime due_date  "borrow_date + 14 days"
    datetime return_date ">= borrow_date"
    decimal fine_amount "<= 20"
    int times_renew "<= 2"
    bool is_returned
  }

  RESERVATION {
    int id PK
    int user_id FK
    int book_id FK
    datetime reservation_date
    datetime expiry_date "reservation_date + 14 days"
    bool is_fulfilled
  }

  FINE {
    int id PK
    int borrow_id FK
    decimal amount "<= 20"
    bool is_paid
    datetime paid_at
  }
